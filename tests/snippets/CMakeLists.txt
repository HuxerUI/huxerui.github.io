cmake_minimum_required(VERSION 3.20)

project(huxerui_documentation_snippets LANGUAGES CXX)

if (NOT HUXERUI_SOURCE_DIR)
    message(FATAL_ERROR "HUXERUI_SOURCE_DIR must point to the documented HuxerUI checkout")
endif ()

add_library(huxerui_documentation_snippets OBJECT docs_snippets.cpp public_api_contract.cpp)
target_compile_features(huxerui_documentation_snippets PRIVATE cxx_std_20)
target_include_directories(huxerui_documentation_snippets PRIVATE "${HUXERUI_SOURCE_DIR}/include")
set(documentation_targets huxerui_documentation_snippets)

if (HUXERUI_DOC_EXAMPLES_DIR)
    file(TO_CMAKE_PATH "${HUXERUI_DOC_EXAMPLES_DIR}" huxerui_doc_examples_dir)
    file(GLOB documentation_examples CONFIGURE_DEPENDS "${huxerui_doc_examples_dir}/*.cpp")
    if (NOT documentation_examples)
        message(FATAL_ERROR "HUXERUI_DOC_EXAMPLES_DIR does not contain extracted C++ examples")
    endif ()
    foreach (example_source IN LISTS documentation_examples)
        get_filename_component(example_name "${example_source}" NAME_WE)
        string(MAKE_C_IDENTIFIER "${example_name}" example_identifier)
        set(example_target "huxerui_documentation_example_${example_identifier}")
        if (TARGET ${example_target})
            message(FATAL_ERROR "Documentation example identifiers collide after CMake normalization: ${example_name}")
        endif ()
        add_library(${example_target} OBJECT "${example_source}")
        target_compile_features(${example_target} PRIVATE cxx_std_20)
        target_include_directories(${example_target} PRIVATE "${HUXERUI_SOURCE_DIR}/include")
        list(APPEND documentation_targets ${example_target})
    endforeach ()
endif ()

foreach (target IN LISTS documentation_targets)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if (MSVC)
        target_compile_options(${target} PRIVATE /wd5030)
    else ()
        target_compile_options(${target} PRIVATE -Wno-attributes)
    endif ()
endforeach ()

option(HUXERUI_DOC_BUILD_MODULES "Compile real import-std documentation examples with MSVC" ${MSVC})
if (HUXERUI_DOC_BUILD_MODULES)
    if (NOT MSVC OR CMAKE_VERSION VERSION_LESS 3.28)
        message(FATAL_ERROR "Module examples require MSVC and CMake 3.28 or later")
    endif ()
    get_filename_component(compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
    get_filename_component(msvc_tools_directory "${compiler_directory}/../../.." ABSOLUTE)
    set(std_module "${msvc_tools_directory}/modules/std.ixx")
    if (NOT EXISTS "${std_module}")
        message(FATAL_ERROR "The selected MSVC toolchain does not provide the std module")
    endif ()
    add_library(huxerui_documentation_std OBJECT)
    target_sources(huxerui_documentation_std PUBLIC FILE_SET CXX_MODULES
            BASE_DIRS "${msvc_tools_directory}/modules" FILES "${std_module}")
    target_compile_features(huxerui_documentation_std PUBLIC cxx_std_23)

    add_library(huxerui_documentation_import OBJECT)
    target_sources(huxerui_documentation_import PUBLIC FILE_SET CXX_MODULES
            BASE_DIRS "${HUXERUI_SOURCE_DIR}/modules" FILES "${HUXERUI_SOURCE_DIR}/modules/huxerui.cppm")
    target_include_directories(huxerui_documentation_import PRIVATE "${HUXERUI_SOURCE_DIR}/include")
    target_compile_features(huxerui_documentation_import PUBLIC cxx_std_23)
    target_compile_options(huxerui_documentation_import PRIVATE /wd5030)

    file(GLOB module_interfaces CONFIGURE_DEPENDS "${huxerui_doc_examples_dir}/modules/*.cppm")
    file(GLOB module_implementations CONFIGURE_DEPENDS "${huxerui_doc_examples_dir}/modules/*.cpp")
    if (NOT module_interfaces OR NOT module_implementations)
        message(FATAL_ERROR "No extracted module application and entry units were found")
    endif ()
    add_library(huxerui_documentation_modules OBJECT ${module_implementations})
    target_sources(huxerui_documentation_modules PUBLIC FILE_SET CXX_MODULES
            BASE_DIRS "${huxerui_doc_examples_dir}/modules" FILES ${module_interfaces})
    target_compile_features(huxerui_documentation_modules PUBLIC cxx_std_23)
    target_link_libraries(huxerui_documentation_modules PRIVATE huxerui_documentation_std huxerui_documentation_import)
else ()
    message(STATUS "Module examples are checked by the separate MSVC module profile")
endif ()
