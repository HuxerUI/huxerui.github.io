cmake_minimum_required(VERSION 3.20)

project(huxerui_documentation_snippets LANGUAGES CXX)

if (NOT HUXERUI_SOURCE_DIR)
    message(FATAL_ERROR "HUXERUI_SOURCE_DIR must point to the documented HuxerUI checkout")
endif ()

add_library(huxerui_documentation_snippets OBJECT docs_snippets.cpp public_api_contract.cpp)
target_compile_features(huxerui_documentation_snippets PRIVATE cxx_std_20)
target_include_directories(huxerui_documentation_snippets PRIVATE "${HUXERUI_SOURCE_DIR}/include")
set(documentation_targets huxerui_documentation_snippets)

if (HUXERUI_DOC_EXAMPLES_DIR)
    file(TO_CMAKE_PATH "${HUXERUI_DOC_EXAMPLES_DIR}" huxerui_doc_examples_dir)
    file(GLOB documentation_examples CONFIGURE_DEPENDS "${huxerui_doc_examples_dir}/*.cpp")
    if (NOT documentation_examples)
        message(FATAL_ERROR "HUXERUI_DOC_EXAMPLES_DIR does not contain extracted C++ examples")
    endif ()
    add_library(huxerui_documentation_examples OBJECT ${documentation_examples})
    target_compile_features(huxerui_documentation_examples PRIVATE cxx_std_20)
    target_include_directories(huxerui_documentation_examples PRIVATE "${HUXERUI_SOURCE_DIR}/include")
    list(APPEND documentation_targets huxerui_documentation_examples)
endif ()

foreach (target IN LISTS documentation_targets)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if (MSVC)
        target_compile_options(${target} PRIVATE /wd5030)
    else ()
        target_compile_options(${target} PRIVATE -Wno-attributes)
    endif ()
endforeach ()
